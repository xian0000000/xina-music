name: Deploy xina music

on:
  push:
    branches: ["main"]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: "pages"
  cancel-in-progress: false

jobs:
  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Buat config.js dari secret
        env:
          FIREBASE_CONFIGS: ${{ secrets.FIREBASE_CONFIGS }}
        run: |
          test -n "$FIREBASE_CONFIGS" || { echo "Secret FIREBASE_CONFIGS belum diisi"; exit 1; }
          printf 'window.FIREBASE_CONFIGS = %s;\n' "$FIREBASE_CONFIGS" > config.js

      - name: Siapkan folder publik
        run: |
          mkdir _site
          cp index.html app.js style.css lame.min.js config.js manifest.webmanifest sw.js _site/
          cp -r icons _site/icons
          cp tick.mp3 _site/ 2>/dev/null || true

      - name: Setup Pages
        uses: actions/configure-pages@v5

      - name: Upload artifact
        uses: actions/upload-pages-artifact@v3
        with:
          path: '_site'

      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v5
